/**
 * =========================================================
 * School Companion - One File Backend + Simple UI
 * =========================================================
 * قبل التشغيل:
 *
 * 1) npm init -y
 * 2) npm install express express-session bcryptjs multer dotenv @supabase/supabase-js
 *
 * 3) أنشئ ملف .env بهذا الشكل:
 *
 * PORT=3000
 * SESSION_SECRET=change_this_secret
 * SUPABASE_URL=https://YOUR_PROJECT.supabase.co
 * SUPABASE_SERVICE_ROLE_KEY=YOUR_SERVICE_ROLE_KEY
 * SUPABASE_BUCKET=school-files
 * CREDLY_URL=https://www.credly.com/users/asim-abdulrahman
 *
 * 4) في Supabase نفّذ هذا SQL مرة واحدة:
 *
 * create table if not exists public.users (
 *   id bigint generated always as identity primary key,
 *   full_name text not null,
 *   email text not null unique,
 *   phone text,
 *   role text not null check (role in ('admin', 'teacher', 'parent')),
 *   password_hash text not null,
 *   created_at timestamptz not null default now()
 * );
 *
 * create table if not exists public.files (
 *   id bigint generated always as identity primary key,
 *   user_id bigint not null references public.users(id) on delete cascade,
 *   role text not null check (role in ('admin', 'teacher', 'parent')),
 *   file_name text not null,
 *   file_path text not null unique,
 *   mime_type text,
 *   size_bytes bigint,
 *   public_url text,
 *   created_at timestamptz not null default now()
 * );
 *
 * 5) أنشئ Bucket في Supabase Storage باسم:
 * school-files
 * واجعله Public إذا أردت روابط مباشرة للملفات.
 *
 * 6) شغّل:
 * node server.js
 *
 * 7) افتح:
 * http://localhost:3000
 */

require("dotenv").config();

function roleLabel(role) {
  if (role === "admin") return "إدارة";
  if (role === "teacher") return "معلم";
  if (role === "parent") return "ولي أمر";
  return role;
}

function escapeHtml(str = "") {
  return String(str)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}

function sanitizeEmail(email) {
  return String(email || "").trim().toLowerCase();
}

function safeFileName(name) {
  return String(name || "file")
    .replace(/[^\w.\-]+/g, "_")
    .replace(/_+/g, "_");
}

function buildStoragePath(role, userId, originalName) {
  return `${role}/${userId}/${Date.now()}_${safeFileName(originalName)}`;
}

function requireAuth(req, res, next) {
  if (!req.session.user) return res.redirect("/login");
  next();
}

function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.session.user) return res.redirect("/login");
    if (!roles.includes(req.session.user.role)) {
      return res
        .status(403)
        .send(
          layout(
            "غير مصرح",
            "<h2>غير مسموح</h2><p>ليس لديك صلاحية لهذه الصفحة.</p>",
            req.session.user
          )
        );
    }
    next();
  };
}

async function getUserByEmail(email) {
  const { data, error } = await supabase
    .from("users")
    .select("*")
    .eq("email", sanitizeEmail(email))
    .maybeSingle();

  if (error) throw error;
  return data;
}

async function getUserById(id) {
  const { data, error } = await supabase
    .from("users")
    .select("id, full_name, email, phone, role, created_at")
    .eq("id", id)
    .maybeSingle();

  if (error) throw error;
  return data;
}

app.post("/register", async (req, res) => {
  try {
    const full_name = String(req.body.full_name || "").trim();
    const email = sanitizeEmail(req.body.email);
    const phone = String(req.body.phone || "").trim() || null;
    const role = String(req.body.role || "").trim();
    const password = String(req.body.password || "");

    if (!full_name || !email || !role || !password) {
      return res.send(
        layout(
          "إنشاء حساب",
          "<h2>بيانات ناقصة</h2><p>أكمل الحقول المطلوبة.</p>",
          null,
          "جميع الحقول الأساسية مطلوبة."
        )
      );
    }

    if (!ROLES.includes(role)) {
      return res.send(
        layout("إنشاء حساب", "<h2>خطأ</h2>", null, "نوع الحساب غير صالح.")
      );
    }

    if (password.length < 6) {
      return res.send(
        layout(
          "إنشاء حساب",
          "<h2>كلمة المرور قصيرة</h2>",
          null,
          "الحد الأدنى 6 أحرف."
        )
      );
    }

    const existing = await getUserByEmail(email);
    if (existing) {
      return res.send(
        layout(
          "إنشاء حساب",
          `<h2>هذا البريد مستخدم بالفعل</h2><a class="btn" href="/login">تسجيل الدخول</a>`,
          null,
          "البريد الإلكتروني موجود مسبقًا."
        )
      );
    }

    const password_hash = await bcrypt.hash(password, 10);

    const { data, error } = await supabase
      .from("users")
      .insert([{ full_name, email, phone, role, password_hash }])
      .select("id, full_name, email, phone, role, created_at")
      .single();

    if (error) throw error;

    req.session.user = data;
    return res.redirect("/dashboard");
  } catch (err) {
    console.error(err);
    return res
      .status(500)
      .send(
        layout(
          "خطأ",
          "<h2>تعذر إنشاء الحساب</h2>",
          null,
          err.message || "حدث خطأ داخلي."
        )
      );
  }
});

app.post("/login", async (req, res) => {
  try {
    const email = sanitizeEmail(req.body.email);
    const password = String(req.body.password || "");

    if (!email || !password) {
      return res.send(
        layout(
          "تسجيل الدخول",
          "<h2>بيانات ناقصة</h2>",
          null,
          "أدخل البريد وكلمة المرور."
        )
      );
    }

    const user = await getUserByEmail(email);
    if (!user) {
      return res.send(
        layout(
          "تسجيل الدخول",
          "<h2>تعذر تسجيل الدخول</h2>",
          null,
          "الحساب غير موجود."
        )
      );
    }

    const ok = await bcrypt.compare(password, user.password_hash);
    if (!ok) {
      return res.send(
        layout(
          "تسجيل الدخول",
          "<h2>تعذر تسجيل الدخول</h2>",
          null,
          "كلمة المرور غير صحيحة."
        )
      );
    }

    req.session.user = {
      id: user.id,
      full_name: user.full_name,
      email: user.email,
      phone: user.phone,
      role: user.role,
      created_at: user.created_at,
    };

    return res.redirect("/dashboard");
  } catch (err) {
    console.error(err);
    return res
      .status(500)
      .send(
        layout(
          "خطأ",
          "<h2>تعذر تسجيل الدخول</h2>",
          null,
          err.message || "حدث خطأ داخلي."
        )
      );
  }
});

app.post("/logout", requireAuth, (req, res) => {
  req.session.destroy((err) => {
    if (err) {
      console.error(err);
      return res
        .status(500)
        .send(
          layout(
            "خطأ",
            "<h2>تعذر تسجيل الخروج</h2>",
            req.session.user,
            "حدث خطأ أثناء تسجيل الخروج."
          )
        );
    }
    res.clearCookie("connect.sid");
    return res.redirect("/login");
  });
});
